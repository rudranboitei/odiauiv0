## Installation

```bash
npx shadcn@latest add https://odiaui.vercel.app/r/accordion.json
```

## Usage

```tsx
import {
  Accordion,
  AccordionContent,
  AccordionItem,
  AccordionTrigger,
} from "@/components/ui/accordion";

export function Example() {
  return (
    <Accordion defaultValue={["item-1"]}>
      <AccordionItem value="item-1">
        <AccordionTrigger>Is it accessible?</AccordionTrigger>
        <AccordionContent>
          Yes. It adheres to the WAI-ARIA design pattern.
        </AccordionContent>
      </AccordionItem>
      <AccordionItem value="item-2">
        <AccordionTrigger>Can I open more than one?</AccordionTrigger>
        <AccordionContent>
          Yes. Add the `multiple` prop to the root to allow several items to
          stay open at once.
        </AccordionContent>
      </AccordionItem>
    </Accordion>
  );
}
```

## Composition

Use the following composition to build an accordion:

```txt
Accordion
├── AccordionItem
│   ├── AccordionTrigger
│   └── AccordionContent
└── AccordionItem
    ├── AccordionTrigger
    └── AccordionContent
```

## Examples

### Odisha

Use `variant="odisha"` for a Konark-inspired look. Expanded cards get a warm
primary gradient, a fading temple accent bar, and a Konark wheel that rotates
into place.

<ComponentPreview name="accordion-odisha">
  <AccordionOdisha />
</ComponentPreview>

### Flush

Use `variant="flush"` for a clean divided list with no card chrome.

<ComponentPreview name="accordion-flush">
  <AccordionFlush />
</ComponentPreview>

### Multiple

Use the `multiple` prop to allow multiple items to be open at the same time.

  <AccordionMultiple />
</ComponentPreview>

### Disabled

Use the `disabled` prop on `AccordionItem` to disable individual items.

## API Reference

### Accordion

| Prop            | Type                                  | Default  |
| --------------- | ------------------------------------- | -------- |
| `variant`       | `"card" \| "flush" \| "odisha"`       | `"card"` |
| `multiple`      | `boolean`                             | `false`  |
| `defaultValue`  | `unknown[]`                           | -        |
| `value`         | `unknown[]`                           | -        |
| `onValueChange` | `(value: unknown[], details) => void` | -        |
| `keepMounted`   | `boolean`                             | `false`  |
| `disabled`      | `boolean`                             | `false`  |
| `className`     | `string`                              | -        |

### AccordionItem

| Prop           | Type                      | Default |
| -------------- | ------------------------- | ------- |
| `value`        | `unknown`                 | auto    |
| `disabled`     | `boolean`                 | `false` |
| `onOpenChange` | `(open: boolean) => void` | -       |
| `className`    | `string`                  | -       |

### AccordionTrigger

| Prop        | Type     | Default |
| ----------- | -------- | ------- |
| `className` | `string` | -       |

### AccordionContent

| Prop        | Type     | Default |
| ----------- | -------- | ------- |
| `className` | `string` | -       |

See the [Base UI documentation](https://base-ui.com/react/components/accordion) for more information.
