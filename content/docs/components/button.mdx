"use client";

import { CheckIcon, ShoppingCartIcon } from "lucide-react";
import * as React from "react";

import { Button } from "./button";

export type AddToCartButtonProps = React.ComponentProps<typeof Button> & {
  onAdd?: () => void;
};

export function AddToCartButton({ onAdd, ...props }: AddToCartButtonProps) {
  const [quantity, setQuantity] = React.useState(0);
  const [justAdded, setJustAdded] = React.useState(false);

  const handleAdd = () => {
    setQuantity((q) => q + 1);
    setJustAdded(true);
    onAdd?.();
    window.setTimeout(() => setJustAdded(false), 1500);
  };

  return (
    <Button onClick={handleAdd} {...props}>
      {justAdded ? (
        <CheckIcon className="size-4" aria-hidden="true" />
      ) : (
        <ShoppingCartIcon className="size-4" aria-hidden="true" />
      )}
      {justAdded ? "Added" : "Add to cart"}
      {quantity > 0 && !justAdded && (
        <span className="inline-flex h-5 min-w-5 items-center justify-center rounded-full bg-primary-foreground/25 px-1 text-xs font-semibold">
          {quantity}
        </span>
      )}
    </Button>
  );
}

"use client";

import { CheckIcon, Loader2Icon, SendIcon } from "lucide-react";
import * as React from "react";

import { Button } from "./button";

export type SendOtpButtonProps = React.ComponentProps<typeof Button> & {
  duration?: number;
  onResend?: () => void | Promise<void>;
};

export function SendOtpButton({
  duration = 30,
  onResend,
  ...props
}: SendOtpButtonProps) {
  const [seconds, setSeconds] = React.useState(0);
  const [sent, setSent] = React.useState(false);

  React.useEffect(() => {
    if (seconds <= 0) return;
    const id = window.setInterval(() => setSeconds((s) => s - 1), 1000);
    return () => window.clearInterval(id);
  }, [seconds]);

  const handleClick = async () => {
    if (seconds > 0) return;
    await onResend?.();
    setSent(true);
    setSeconds(duration);
    window.setTimeout(() => setSent(false), 1500);
  };

  const disabled = seconds > 0 || props.disabled;

  return (
    <Button
      onClick={handleClick}
      disabled={disabled}
      aria-busy={props["aria-busy"]}
      {...props}
    >
      {sent ? (
        <CheckIcon className="size-4" aria-hidden="true" />
      ) : seconds > 0 ? (
        <Loader2Icon className="size-4 animate-spin" aria-hidden="true" />
      ) : (
        <SendIcon className="size-4" aria-hidden="true" />
      )}
      {seconds > 0 ? `Resend OTP in ${seconds}s` : sent ? "Sent" : "Send OTP"}
    </Button>
  );
}

"use client";

import { Share2Icon } from "lucide-react";
import * as React from "react";

import { Button } from "./button";

export type WhatsappShareButtonProps = React.ComponentProps<typeof Button> & {
  text?: string;
  url?: string;
};

export function WhatsappShareButton({
  text = "",
  url,
  children = "Share on WhatsApp",
  onClick,
  ...props
}: WhatsappShareButtonProps) {
  const handleClick: React.MouseEventHandler<HTMLButtonElement> = (e) => {
    onClick?.(e);
    const shareUrl =
      url ?? (typeof window !== "undefined" ? window.location.href : "");
    const message = text ? `${text} ${shareUrl}` : shareUrl;
    window.open(
      `https://wa.me/?text=${encodeURIComponent(message)}`,
      "_blank",
      "noopener,noreferrer"
    );
  };

  return (
    <Button
      variant="secondary"
      aria-label="Share on WhatsApp"
      onClick={handleClick}
      {...props}
    >
      <Share2Icon className="size-4" aria-hidden="true" />
      {children}
    </Button>
  );
}

"use client";

import { HeartIcon } from "lucide-react";
import { motion } from "motion/react";
import * as React from "react";

import { cn } from "@/lib/utils";
import { Button } from "./button";

export type WishlistButtonProps = React.ComponentProps<typeof Button> & {
  defaultLiked?: boolean;
  liked?: boolean;
  onLikedChange?: (liked: boolean) => void;
};

export function WishlistButton({
  defaultLiked = false,
  liked: likedProp,
  onLikedChange,
  className,
  ...props
}: WishlistButtonProps) {
  const [uncontrolled, setUncontrolled] = React.useState(defaultLiked);
  const liked = likedProp ?? uncontrolled;

  const toggle = () => {
    const next = !liked;
    if (likedProp === undefined) setUncontrolled(next);
    onLikedChange?.(next);
  };

  return (
    <Button
      aria-pressed={liked}
      aria-label={liked ? "Remove from wishlist" : "Add to wishlist"}
      className={cn("gap-1.5", className)}
      onClick={toggle}
      variant={liked ? "secondary" : "outline"}
      {...props}
    >
      <motion.span
        animate={{ scale: liked ? [1, 1.35, 1] : 1 }}
        className="inline-flex"
        transition={{ duration: 0.3 }}
      >
        <HeartIcon
          className={cn("size-4", liked && "fill-current text-red-500")}
          aria-hidden="true"
        />
      </motion.span>
      {liked ? "Saved" : "Wishlist"}
    </Button>
  );
}

"use client";

import { CheckIcon, CopyIcon } from "lucide-react";
import * as React from "react";

import { Button } from "./button";

export type CopyButtonProps = React.ComponentProps<typeof Button> & {
  value: string;
  copiedLabel?: string;
};

export function CopyButton({
  value,
  copiedLabel = "Copied",
  children = "Copy",
  onClick,
  ...props
}: CopyButtonProps) {
  const [copied, setCopied] = React.useState(false);

  const handleClick: React.MouseEventHandler<HTMLButtonElement> = async (e) => {
    onClick?.(e);
    try {
      await navigator.clipboard.writeText(value);
      setCopied(true);
      window.setTimeout(() => setCopied(false), 2000);
    } catch {
      // ignore
    }
  };

  return (
    <Button variant="outline" onClick={handleClick} {...props}>
      {copied ? (
        <CheckIcon className="size-4" aria-hidden="true" />
      ) : (
        <CopyIcon className="size-4" aria-hidden="true" />
      )}
      {copied ? copiedLabel : children}
    </Button>
  );
}

import { CheckIcon, DownloadIcon, Loader2Icon } from "lucide-react";
import * as React from "react";

import { Button } from "./button";

export type DownloadButtonProps = React.ComponentProps<typeof Button> & {
  progress?: number;
};

export function DownloadButton({
  progress,
  children = "Download",
  onClick,
  disabled,
  ...props
}: DownloadButtonProps) {
  const [downloading, setDownloading] = React.useState(false);
  const [done, setDone] = React.useState(false);

  const handleClick: React.MouseEventHandler<HTMLButtonElement> = async (e) => {
    onClick?.(e);
    if (progress === undefined) {

      window.setTimeout(() => {
        setDownloading(false);
        setDone(true);
        window.setTimeout(() => setDone(false), 2000);
      }, 1200);
    }
  };

  const isLoading = downloading || (progress !== undefined && progress < 100 && progress > 0);
  const isDone = done || progress === 100;

  return (
    <Button
      onClick={handleClick}
      disabled={disabled || isLoading}
      aria-busy={isLoading || undefined}
      {...props}

      {isDone ? (
        <CheckIcon className="size-4" aria-hidden="true" />
      ) : isLoading ? (
        <Loader2Icon className="size-4 animate-spin" aria-hidden="true" />
      ) : (
        <DownloadIcon className="size-4" aria-hidden="true" />
      )}
      {isDone ? "Downloaded" : isLoading && progress !== undefined ? `Downloading ${Math.round(progress)}%` : children}
    </Button>
  );
}

"use client";

import { CheckIcon, TrashIcon } from "lucide-react";
import * as React from "react";

import { Button } from "./button";

export type DeleteConfirmButtonProps = React.ComponentProps<typeof Button> & {
  onConfirm?: () => void;
  onDelete?: () => void;
};

export function DeleteConfirmButton({
  onConfirm,
  onDelete,
  children = "Delete",
  ...props
}: DeleteConfirmButtonProps) {
  const [armed, setArmed] = React.useState(false);
  const [done, setDone] = React.useState(false);

  const handleClick = () => {
    if (done) return;
    if (armed) {
      (onConfirm ?? onDelete)?.();
      setArmed(false);
      setDone(true);
      window.setTimeout(() => setDone(false), 2000);
    } else {
      setArmed(true);
      window.setTimeout(() => setArmed(false), 3000);
    }
  };

  let label: React.ReactNode = children;
  if (armed) label = "Are you sure?";
  if (done) label = "Deleted";

  return (
    <Button
      onClick={handleClick}
      variant={armed || done ? "destructive" : "outline"}
      {...props}
    >
      {done ? (
        <CheckIcon className="size-4" aria-hidden="true" />
      ) : (
        <TrashIcon className="size-4" aria-hidden="true" />
      )}
      {label}
    </Button>
  );
}

"use client";

import * as React from "react";

import { Button } from "./button";

export type LanguageToggleButtonProps = React.ComponentProps<typeof Button> & {
  value?: "or" | "en";
  onChange?: (value: "or" | "en") => void;
};

export function LanguageToggleButton({
  value,
  onChange,
  onClick,
  children,
  ...props
}: LanguageToggleButtonProps) {
  const [internal, setInternal] = React.useState<"or" | "en">(value ?? "or");
  const current = value ?? internal;

  const handleClick: React.MouseEventHandler<HTMLButtonElement> = (e) => {
    onClick?.(e);
    const next = current === "or" ? "en" : "or";
    if (value === undefined) setInternal(next);
    onChange?.(next);
  };

  return (
    <Button
      variant="outline"
      aria-label="Toggle language"
      onClick={handleClick}
      {...props}
    >
      {children ?? (current === "or" ? "ଓଡ଼ିଆ / EN" : "EN / ଓଡ଼ିଆ")}
    </Button>
  );
}
